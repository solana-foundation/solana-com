  getU64Decoder,

/** Pyth's receiver program, which owns every price feed account it writes. */
export const PYTH_RECEIVER_PROGRAM = address(
  "rec5EKMGg6MxZYaMdyBfgwp4d5rB9T1VQH5pJv5LtFJ",

// First 8 bytes of every PriceUpdateV2 account (an Anchor discriminator).
const PRICE_UPDATE_V2 = "22f123639d7ef4cd";

const hex = (bytes: Uint8Array) => Buffer.from(bytes).toString("hex");

// The price message inside a PriceUpdateV2 account.
const priceMessage = getStructDecoder([
  ["feedId", fixDecoderSize(getBytesDecoder(), 32)],
  ["price", getI64Decoder()],
  ["conf", getU64Decoder()],
  ["exponent", getI32Decoder()],
  ["publishTime", getI64Decoder()],
]);

/**
 * Read a Pyth price feed account and refuse a price you shouldn't trade on:
 * the wrong account, the wrong feed, partly verified, too old, or with a
 * confidence interval that is too wide.
 */
export async function readPrice(
  feed: Address = USDC_USD_FEED,
  {
    feedId = USDC_USD_FEED_ID,
    maxAgeSeconds = Number(process.env.MAX_PRICE_AGE_SECONDS ?? 120),
    maxConfidenceBps = 50n, // 0.5% of the price
  } = {},
) {
  const account = await fetchEncodedAccount(rpc, feed);
  if (!account.exists) throw new Error(`no price feed at ${feed}`);
  if (account.programAddress !== PYTH_RECEIVER_PROGRAM)
    throw new Error(`${feed} is not owned by the Pyth receiver program`);
  const data = account.data;
  if (hex(data.slice(0, 8)) !== PRICE_UPDATE_V2)
    throw new Error(`${feed} is not a PriceUpdateV2 account`);
  // Byte 40 is the verification level: 1 = Full (all guardian signatures).
  if (data[40] !== 1) throw new Error("price update is only partly verified");

  const m = priceMessage.decode(data, 41);
  if (hex(new Uint8Array(m.feedId)) !== feedId)
    throw new Error(`${feed} holds a different feed`);
  const ageSeconds = Math.floor(Date.now() / 1000) - Number(m.publishTime);
  if (ageSeconds > maxAgeSeconds)
    throw new Error(`price is ${ageSeconds} s old (max ${maxAgeSeconds} s)`);
  if (m.price <= 0n || m.conf * 10_000n > m.price * maxConfidenceBps)
    throw new Error("price confidence interval is too wide");

  return {
    price: m.price, // the price is price × 10^exponent
    conf: m.conf,
    exponent: m.exponent,
    publishTime: m.publishTime,
    ageSeconds,
  };
}

/** Show a raw value (value × 10^exponent) as a decimal string. */
export function decimal(value: bigint, exponent: number) {
  const s = value.toString().padStart(1 - exponent, "0");
  return `${s.slice(0, exponent)}.${s.slice(exponent)}`;
}
/** 6-decimal base units as a plain number, e.g. 4000000n -> "4". */
export const units = (n: bigint) => decimal(n, -6).replace(/\.?0+$/, "");
const usd = (n: bigint) => `$${decimal(n / 10_000n, -2)}`;

/**
 * Price the cash leg. The asset is quoted in US dollars per share; the buyer
 * pays in USDC, so convert the dollar amount at the oracle's USDC/USD price.
 * Amounts are in base units: LSF and USDC both have 6 decimals.
 */
export async function priceTrade(shares: bigint, navPerShareUsd: bigint) {
  const usdc = await readPrice();
  const notionalUsd = (shares * navPerShareUsd) / 1_000_000n;
  // USDC = USD / (price × 10^exponent), rounded up to the next base unit.
  const scale = 10n ** BigInt(-usdc.exponent);
  const cashAmount = (notionalUsd * scale + usdc.price - 1n) / usdc.price;
  console.log(
    "USDC/USD from Pyth:",
    decimal(usdc.price, usdc.exponent),
    "±",
    decimal(usdc.conf, usdc.exponent),
    `(published ${usdc.ageSeconds} s ago)`,
  );
  console.log(
    `Price: ${units(shares)} LSF × ${usd(navPerShareUsd)} = ${usd(notionalUsd)} = ${units(cashAmount)} USDC`,
  );
  return { usdc, notionalUsd, assetAmount: shares, cashAmount };
}
