```ts
// npx tsx inspect-mint.ts
import { address, type Option } from "@solana/kit";
import { fetchMint } from "@solana-program/token-2022";
import { fetchMaybeMintConfig, findMintConfigPda } from "@solana/token-acl-sdk";
import { env, rpc } from "./common";

const mint = address(env("MINT"));
const unwrap = <T>(option: Option<T>) =>
  option.__option === "Some" ? option.value : null;

// The mint account carries the extensions. The Token ACL config is a separate
// account, derived from the mint, that holds the gate program and the
// permissionless-thaw flag.
const { data } = await fetchMint(rpc, mint, { commitment: "confirmed" });
const [mintConfigAddress] = await findMintConfigPda({ mint });
const mintConfig = await fetchMaybeMintConfig(rpc, mintConfigAddress);
const extensions = unwrap(data.extensions) ?? [];

console.log({
  mint,
  supply: data.supply.toString(),
  decimals: data.decimals,
  mintAuthority: unwrap(data.mintAuthority),
  freezeAuthority: unwrap(data.freezeAuthority),
  tokenAcl:
    mintConfig.exists && unwrap(data.freezeAuthority) === mintConfigAddress,
  gatingProgram: mintConfig.exists ? mintConfig.data.gatingProgram : null,
  permissionlessThaw: mintConfig.exists
    ? mintConfig.data.enablePermissionlessThaw
    : false,
  extensions: extensions.map((ext) => ext.__kind),
});
```

```ts
// npx tsx inspect-mint.ts
import type { Option } from "@solana/kit";
import { PublicKey } from "@solana/web3.js";
import { getMintDecoder } from "@solana-program/token-2022";
import { findMintConfigPda, getMintConfigDecoder } from "@solana/token-acl-sdk";
import { addressOf, connection, env } from "./common";

const mint = addressOf(env("MINT"));
const unwrap = <T>(option: Option<T>) =>
  option.__option === "Some" ? option.value : null;

// The mint account carries the extensions. The Token ACL config is a separate
// account, derived from the mint, that holds the gate program and the
// permissionless-thaw flag.
const mintAccount = await connection.getAccountInfo(new PublicKey(mint));
const data = getMintDecoder().decode(mintAccount!.data);
const [mintConfigAddress] = await findMintConfigPda({ mint });
const configAccount = await connection.getAccountInfo(
  new PublicKey(mintConfigAddress),
);
const mintConfig = configAccount
  ? getMintConfigDecoder().decode(configAccount.data)
  : null;
const extensions = unwrap(data.extensions) ?? [];

console.log({
  mint,
  supply: data.supply.toString(),
  decimals: data.decimals,
  mintAuthority: unwrap(data.mintAuthority),
  freezeAuthority: unwrap(data.freezeAuthority),
  tokenAcl:
    mintConfig !== null && unwrap(data.freezeAuthority) === mintConfigAddress,
  gatingProgram: mintConfig?.gatingProgram ?? null,
  permissionlessThaw: mintConfig?.enablePermissionlessThaw ?? false,
  extensions: extensions.map((ext) => ext.__kind),
});
```

```rust
// cargo run --release --bin inspect-mint
use anyhow::Result;
use gated_token::{acl, env, rpc};
use solana_sdk::pubkey::Pubkey;
use spl_token_2022_interface::{
    extension::{BaseStateWithExtensions, StateWithExtensions},
    state::Mint,
};

#[tokio::main]
async fn main() -> Result<()> {
    let rpc = rpc();
    let mint = env("MINT")?;

    // The mint account carries the extensions. The Token ACL config is a
    // separate account, derived from the mint: discriminator (1), bump (1),
    // thaw flag (1), freeze flag (1), mint (32), authority (32), gate (32).
    let data = rpc.get_account_data(&mint).await?;
    let state = StateWithExtensions::<Mint>::unpack(&data)?;
    let config_address = acl::mint_config(&mint);
    let config = rpc
        .get_account_with_commitment(&config_address, rpc.commitment())
        .await?
        .value;
    let mint_authority: Option<Pubkey> = state.base.mint_authority.into();
    let freeze_authority: Option<Pubkey> = state.base.freeze_authority.into();

    println!("mint: {mint}");
    println!("supply: {}", state.base.supply);
    println!("decimals: {}", state.base.decimals);
    println!("mintAuthority: {mint_authority:?}");
    println!("freezeAuthority: {freeze_authority:?}");
    println!(
        "tokenAcl: {}",
        config.is_some() && freeze_authority == Some(config_address)
    );
    if let Some(config) = &config {
        println!(
            "gatingProgram: {}",
            Pubkey::try_from(&config.data[68..100])?
        );
        println!("permissionlessThaw: {}", config.data[2] == 1);
    }
    println!("extensions: {:?}", state.get_extension_types()?);
    Ok(())
}
```
