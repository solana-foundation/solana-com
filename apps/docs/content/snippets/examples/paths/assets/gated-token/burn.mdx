const [account] = await findAssociatedTokenPda({
  mint,
  owner: holder.address,
  tokenProgram: TOKEN_2022_PROGRAM_ADDRESS,
});

// Token-2022 rejects a plain burn on this mint. The permissioned burn
// instruction carries both signers: the holder and the burn authority.
await send(holder, [
  getPermissionedBurnCheckedInstruction({
    account,
    mint,
    permissionedBurnAuthority: burnAuthority,
    authority: holder,
    amount: toBaseUnits(arg(1, "the amount"), data.decimals),

]);

} from "@solana-program/token-2022";

  addressOf,
  arg,
  connection,
  env,
  loadKeypair,
  send,
  toBaseUnits,
} from "./common";

const holder = await loadKeypair("demo-holder");
const burnAuthority = await loadKeypair("demo-authorities/burn");
const mint = addressOf(env("MINT"));
const mintAccount = await connection.getAccountInfo(new PublicKey(mint));
const { decimals } = getMintDecoder().decode(mintAccount!.data);
const [account] = await findAssociatedTokenPda({
  mint,
  owner: holder.publicKey.toBase58(),
  tokenProgram: TOKEN_2022_PROGRAM_ADDRESS,
});

// Token-2022 rejects a plain burn on this mint. The permissioned burn
// instruction carries both signers: the holder and the burn authority.
await send(
  [holder, burnAuthority],
  [
    getPermissionedBurnCheckedInstruction({
      account,
      mint,
      permissionedBurnAuthority: burnAuthority,
      authority: holder,
      amount: toBaseUnits(arg(1, "the amount"), decimals),
      decimals,
    }),
  ],
);

// cargo run --release --bin burn -- <amount>
use anyhow::Result;
use gated_token::{
    arg, ata, env, load_keypair, mint_decimals, rpc, send, to_base_units, TOKEN_2022,
};
use solana_sdk::signature::Signer;
use spl_token_2022_interface::extension::permissioned_burn::instruction::burn_checked;

#[tokio::main]
async fn main() -> Result<()> {
    let rpc = rpc();
    let holder = load_keypair("demo-holder")?;
    let burn_authority = load_keypair("demo-authorities/burn")?;
    let mint = env("MINT")?;
    let decimals = mint_decimals(&rpc, &mint).await?;

    // Token-2022 rejects a plain burn on this mint. The permissioned burn
    // instruction carries both signers: the holder and the burn authority.
    let instruction = burn_checked(
        &TOKEN_2022,
        &ata(&mint, &holder.pubkey()),
        &mint,
        &burn_authority.pubkey(),
        &holder.pubkey(),
        &[],
        to_base_units(&arg(1, "the amount")?, decimals)?,
        decimals,
    )?;
    send(&rpc, &[&holder, &burn_authority], &[instruction]).await
}
