```py Python
import struct

from solana.rpc.api import Client
from solders.pubkey import Pubkey

METADATA_PROGRAM_ID = Pubkey.from_string("metaqbxxUerdq28cj1RbAWkYQm3ybzjb6a8bt518x1s")

client = Client("http://localhost:8899")

# Example: USDC mint address
mint = Pubkey.from_string("EPjFWdd5AufqSSqeM2qN1xzybapC8G4wEGGkZwyTDt1v")

# The metadata account is a PDA derived from the mint address
metadata_address, _ = Pubkey.find_program_address(
    [b"metadata", bytes(METADATA_PROGRAM_ID), bytes(mint)], METADATA_PROGRAM_ID
)
account = client.get_account_info(metadata_address).value

if account is None:
    print("No metadata found for this mint")
else:
    # Layout: key (1 byte), update authority (32), mint (32), then name, symbol, uri
    data = account.data
    fields, offset = [], 65
    for _ in range(3):
        (length,) = struct.unpack_from("<I", data, offset)
        fields.append(data[offset + 4 : offset + 4 + length].decode().rstrip("\x00"))
        offset += 4 + length
    name, symbol, uri = fields

    print("Token Name:", name)
    print("Token Symbol:", symbol)
    print("Metadata URI:", uri)
    print("Update Authority:", Pubkey.from_bytes(data[1:33]))
```

```go Go
package main

import (
	"context"
	"errors"
	"fmt"
	"strings"

	bin "github.com/gagliardetto/binary"
	"github.com/solana-foundation/solana-go/v2"
	"github.com/solana-foundation/solana-go/v2/rpc"
)

func main() {
	client := rpc.New("http://localhost:8899")

	// Example: USDC mint address
	mint := solana.MustPublicKeyFromBase58("EPjFWdd5AufqSSqeM2qN1xzybapC8G4wEGGkZwyTDt1v")

	// The metadata account is a PDA derived from the mint address
	metadataAddress, _, err := solana.FindTokenMetadataAddress(mint)
	if err != nil {
		panic(err)
	}
	account, err := client.GetAccountInfo(context.Background(), metadataAddress)
	if errors.Is(err, rpc.ErrNotFound) {
		fmt.Println("No metadata found for this mint")
		return
	}
	if err != nil {
		panic(err)
	}

	// The first fields of the Metaplex Metadata account
	var metadata struct {
		Key             uint8
		UpdateAuthority solana.PublicKey
		Mint            solana.PublicKey
		Name            string
		Symbol          string
		Uri             string
	}
	if err := bin.NewBorshDecoder(account.GetBinary()).Decode(&metadata); err != nil {
		panic(err)
	}

	fmt.Println("Token Name:", strings.TrimRight(metadata.Name, "\x00"))
	fmt.Println("Token Symbol:", strings.TrimRight(metadata.Symbol, "\x00"))
	fmt.Println("Metadata URI:", strings.TrimRight(metadata.Uri, "\x00"))
	fmt.Println("Update Authority:", metadata.UpdateAuthority)
}
```
