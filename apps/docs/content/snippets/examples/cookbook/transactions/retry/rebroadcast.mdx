```ts title="Web3.js (legacy)"
import {
  Keypair,
  Connection,
  LAMPORTS_PER_SOL,
  SystemProgram,
  Transaction
} from "@solana/web3.js";
import nacl from "tweetnacl";

const sleep = async (ms: number) => {
  return new Promise((r) => setTimeout(r, ms));
};

(async () => {
  const payer = Keypair.generate();
  const toAccount = Keypair.generate().publicKey;

  const connection = new Connection("http://127.0.0.1:8899", "confirmed");

  const airdropSignature = await connection.requestAirdrop(
    payer.publicKey,
    LAMPORTS_PER_SOL
  );
  await connection.confirmTransaction({
    signature: airdropSignature,
    ...(await connection.getLatestBlockhash())
  });

  // The transaction can land until the block height passes lastValidBlockHeight
  const { blockhash, lastValidBlockHeight } =
    await connection.getLatestBlockhash();

  const transaction = new Transaction({
    feePayer: payer.publicKey,
    blockhash,
    lastValidBlockHeight
  }).add(
    SystemProgram.transfer({
      fromPubkey: payer.publicKey,
      toPubkey: toAccount,
      lamports: 1000000
    })
  );
  const message = transaction.serializeMessage();
  const signature = nacl.sign.detached(message, payer.secretKey);
  transaction.addSignature(payer.publicKey, Buffer.from(signature));
  const rawTransaction = transaction.serialize();

  // maxRetries: 0 turns off the RPC node's own rebroadcasting
  const txid = await connection.sendRawTransaction(rawTransaction, {
    skipPreflight: true,
    maxRetries: 0
  });

  // Resend the same signed transaction every 500 ms until it is confirmed or
  // its blockhash expires
  let blockheight = await connection.getBlockHeight();
  while (blockheight <= lastValidBlockHeight) {
    await sleep(500);
    const { value: status } = await connection.getSignatureStatus(txid);
    // A "processed" transaction can still be dropped, so wait for "confirmed"
    if (
      status?.confirmationStatus === "confirmed" ||
      status?.confirmationStatus === "finalized"
    ) {
      if (status.err) {
        throw new Error(`Transaction failed: ${JSON.stringify(status.err)}`);
      }
      console.log("Signature:", txid);
      return;
    }
    await connection
      .sendRawTransaction(rawTransaction, { skipPreflight: true, maxRetries: 0 })
      .catch(() => {}); // a node may reject a copy it already processed
    blockheight = await connection.getBlockHeight();
  }
  throw new Error("Blockhash expired: sign the transaction again");
})();
```

```py Python
import asyncio

from solana.rpc.async_api import AsyncClient
from solana.rpc.commitment import Confirmed
from solana.rpc.models import TxOpts
from solders.keypair import Keypair
from solders.message import MessageV0
from solders.system_program import TransferParams, transfer
from solders.transaction import VersionedTransaction
from solders.transaction_status import TransactionConfirmationStatus


async def main():
    async with AsyncClient("http://localhost:8899", commitment=Confirmed) as client:
        payer = Keypair()
        to_account = Keypair().pubkey()
        airdrop = await client.request_airdrop(payer.pubkey(), 1_000_000_000)
        await client.confirm_transaction(airdrop.value)

        latest = (await client.get_latest_blockhash()).value
        transfer_instruction = transfer(
            TransferParams(
                from_pubkey=payer.pubkey(), to_pubkey=to_account, lamports=1_000_000
            )
        )
        message = MessageV0.try_compile(
            payer.pubkey(), [transfer_instruction], [], latest.blockhash
        )
        transaction = VersionedTransaction(message, [payer])
        raw_transaction = bytes(transaction)
        signature = transaction.signatures[0]

        # Resend the same signed transaction until it lands or its blockhash expires.
        # max_retries=0 turns off the RPC node's own rebroadcasting.
        while (await client.get_block_height()).value <= latest.last_valid_block_height:
            await client.send_raw_transaction(
                raw_transaction, opts=TxOpts(skip_preflight=True, max_retries=0)
            )
            await asyncio.sleep(0.5)
            status = (await client.get_signature_statuses([signature])).value[0]
            # Keep resending until confirmed: a "processed" transaction can still be dropped.
            if (
                status is not None
                and status.confirmation_status != TransactionConfirmationStatus.Processed
            ):
                if status.err:
                    raise RuntimeError(f"Transaction failed: {status.err}")
                break

        print("Signature:", signature)


asyncio.run(main())
```

```go Go
package main

import (
	"context"
	"fmt"
	"time"

	"github.com/solana-foundation/solana-go/v2"
	"github.com/solana-foundation/solana-go/v2/programs/system"
	"github.com/solana-foundation/solana-go/v2/rpc"
)

func main() {
	ctx := context.Background()
	client := rpc.New("http://localhost:8899")

	payer := solana.NewWallet()
	toAccount := solana.NewWallet().PublicKey()
	_, err := client.RequestAirdrop(ctx, payer.PublicKey(), solana.LAMPORTS_PER_SOL, rpc.CommitmentConfirmed)
	if err != nil {
		panic(err)
	}
	time.Sleep(2 * time.Second)

	latest, err := client.GetLatestBlockhash(ctx, rpc.CommitmentConfirmed)
	if err != nil {
		panic(err)
	}
	tx, err := solana.NewTransaction(
		[]solana.Instruction{
			system.NewTransferInstruction(1_000_000, payer.PublicKey(), toAccount).Build(),
		},
		latest.Value.Blockhash,
		solana.TransactionPayer(payer.PublicKey()),
	)
	if err != nil {
		panic(err)
	}
	tx.Sign(func(solana.PublicKey) *solana.PrivateKey { return &payer.PrivateKey })
	rawTx, err := tx.MarshalBinary()
	if err != nil {
		panic(err)
	}
	signature := tx.Signatures[0]

	// Resend the same signed transaction until it lands or its blockhash expires.
	// MaxRetries 0 turns off the RPC node's own rebroadcasting.
	maxRetries := uint(0)
	for {
		height, err := client.GetBlockHeight(ctx, rpc.CommitmentConfirmed)
		if err != nil {
			panic(err)
		}
		if height > latest.Value.LastValidBlockHeight {
			break
		}
		client.SendRawTransactionWithOpts(ctx, rawTx, rpc.TransactionOpts{SkipPreflight: true, MaxRetries: &maxRetries})
		time.Sleep(500 * time.Millisecond)
		statuses, err := client.GetSignatureStatuses(ctx, false, signature)
		if err != nil {
			panic(err)
		}
		// Keep resending until confirmed: a "processed" transaction can still be dropped.
		if status := statuses.Value[0]; status != nil && status.ConfirmationStatus != rpc.ConfirmationStatusProcessed {
			if status.Err != nil {
				panic(fmt.Sprintf("transaction failed: %v", status.Err))
			}
			break
		}
	}

	fmt.Println("Signature:", signature)
}
```
