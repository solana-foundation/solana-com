function withDontFront(ix: Instruction): Instruction {
  return {
    ...ix,
    accounts: [
      ...(ix.accounts ?? []),
      { address: DONT_FRONT, role: AccountRole.READONLY },

use solana_client::nonblocking::rpc_client::RpcClient;
use solana_commitment_config::CommitmentConfig;

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    let rpc = RpcClient::new_with_commitment(
        "http://localhost:8899".to_string(),
        CommitmentConfig::confirmed(),
    );

    let sender = Keypair::new();
    let recipient = Keypair::new();
    let dontfront = Pubkey::from_str(DONT_FRONT)?;

    // Fund the sender so the tx is signable
    let airdrop = rpc
        .request_airdrop(&sender.pubkey(), LAMPORTS_PER_SOL)
        .await?;
    loop {
        if rpc.confirm_transaction(&airdrop).await? {
            break;
        }
    }

    // Build a transfer and append the dontfront account as read-only
    let mut transfer_ix = transfer(&sender.pubkey(), &recipient.pubkey(), 1_000_000);
    transfer_ix
        .accounts
        .push(AccountMeta::new_readonly(dontfront, false));

    // Jito tip — SOL transfer to a random tip account (min 1000 lamports)
    let tip_idx = (std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)?
        .subsec_nanos() as usize)
        % TIP_ACCOUNTS.len();
    let tip_account = Pubkey::from_str(TIP_ACCOUNTS[tip_idx])?;
    let tip_ix = transfer(&sender.pubkey(), &tip_account, 1_000);

    let recent_blockhash = rpc.get_latest_blockhash().await?;

    let transaction = Transaction::new_signed_with_payer(
        &[transfer_ix, tip_ix],
        Some(&sender.pubkey()),
        &[&sender],
        recent_blockhash,
    );

    // Serialize → base64
    let serialized = bincode::serialize(&transaction)?;
    let base64_tx = STANDARD.encode(&serialized);

    // Sending through Jito submits a real mainnet transaction, so it is off by
    // default and the example prints the signed transaction instead. To send it,
    // sign with a funded mainnet keypair, take the blockhash from a mainnet RPC
    // node, and set SEND_TO_JITO=1.
    if std::env::var("SEND_TO_JITO").as_deref() == Ok("1") {
        // Send via Jito's block engine, not a regular RPC node.
        let response = reqwest::Client::new()
            .post(JITO_ENDPOINT)
            .json(&serde_json::json!({
                "jsonrpc": "2.0",
                "id": 1,
                "method": "sendTransaction",
                "params": [base64_tx, {"encoding": "base64"}]
            }))
            .send()
            .await?;
        let status = response.status();
        let body: serde_json::Value = response.json().await?;
        match body["result"].as_str() {
            Some(signature) if status.is_success() && body.get("error").is_none() => {
                println!("Sent with dontfront protection: {signature}")
            }
            _ => anyhow::bail!("Jito rejected the transaction (HTTP {status}): {body}"),
        }
    } else {
        println!("Signed transaction (base64): {base64_tx}");
    }

    Ok(())
}

    "3AVi9Tg9Uo68tJfuvoKvqKNWKkC5wPdSSdeBnizKZ6jT",
]


async def main():
    async with AsyncClient("http://localhost:8899", commitment=Confirmed) as client:
        sender = Keypair()
        recipient = Keypair()

        airdrop = await client.request_airdrop(sender.pubkey(), 1_000_000_000)
        await client.confirm_transaction(airdrop.value)

        # Append jitodontfront as a read-only account to the transfer instruction.
        transfer_ix = transfer(
            TransferParams(
                from_pubkey=sender.pubkey(), to_pubkey=recipient.pubkey(), lamports=1_000_000
            )
        )
        transfer_ix = Instruction(
            transfer_ix.program_id,
            transfer_ix.data,
            transfer_ix.accounts + [AccountMeta(DONT_FRONT, is_signer=False, is_writable=False)],
        )

        # Jito tip: at least 1000 lamports to a random tip account.
        tip_ix = transfer(
            TransferParams(
                from_pubkey=sender.pubkey(),
                to_pubkey=Pubkey.from_string(random.choice(TIP_ACCOUNTS)),
                lamports=1_000,
            )
        )

        blockhash = (await client.get_latest_blockhash()).value.blockhash
        message = MessageV0.try_compile(sender.pubkey(), [transfer_ix, tip_ix], [], blockhash)
        transaction = VersionedTransaction(message, [sender])
        base64_tx = base64.b64encode(bytes(transaction)).decode()

    # Sending through Jito submits a real mainnet transaction, so it is off by
    # default and the example prints the signed transaction instead. To send it,
    # sign with a funded mainnet keypair, take the blockhash from a mainnet RPC
    # node, and set SEND_TO_JITO=1.
    if os.environ.get("SEND_TO_JITO") != "1":
        print("Signed transaction (base64):", base64_tx)
        return

    # Send through Jito's block engine, not a regular RPC node.
    request = urllib.request.Request(
        JITO_ENDPOINT,
        data=json.dumps(
            {
                "jsonrpc": "2.0",
                "id": 1,
                "method": "sendTransaction",
                "params": [base64_tx, {"encoding": "base64"}],
            }
        ).encode(),
        headers={"Content-Type": "application/json"},
    )
    try:
        with urllib.request.urlopen(request) as response:
            result = json.load(response)
    except urllib.error.HTTPError as error:
        raise RuntimeError(f"Jito rejected the transaction: {error.read().decode()}") from error
    if "error" in result:
        raise RuntimeError(f"Jito rejected the transaction: {result['error']}")
    print("Sent with dontfront protection:", result["result"])


asyncio.run(main())

package main

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"

const jitoEndpoint = "https://mainnet.block-engine.jito.wtf/api/v1/transactions"

// Any pubkey starting with "jitodontfront"; it does not need to exist onchain.
var dontFront = solana.MustPublicKeyFromBase58("jitodontfront111111111111111111111111111111")

// Jito tip accounts: https://docs.jito.wtf/lowlatencytxnsend/#gettipaccounts
var tipAccounts = []string{
	"96gYZGLnJYVFmbjzopPSU6QiEV5fGqZNyN9nmNhvrZU5",
	"HFqU5x63VTqvQss8hp11i4wVV8bD44PvwucfZ2bU7gRe",
	"Cw8CFyM9FkoMi7K7Crf6HNQqf4uEMzpKw6QNghXLvLkY",

	"ADuUkR4vqLUMWXxW9gh6D6L8pMSawimctcNZ5pGwDcEt",
	"DttWaMuVvTiduZRnguLF7jNxTgiMBZ1hyAumKUiL2KRL",
	"3AVi9Tg9Uo68tJfuvoKvqKNWKkC5wPdSSdeBnizKZ6jT",
}

func main() {
	ctx := context.Background()
	client := rpc.New("http://localhost:8899")

	sender := solana.NewWallet()
	recipient := solana.NewWallet()

	_, err := client.RequestAirdrop(ctx, sender.PublicKey(), solana.LAMPORTS_PER_SOL, rpc.CommitmentConfirmed)
	if err != nil {
		panic(err)
	}
	time.Sleep(2 * time.Second)

	// Append jitodontfront as a read-only account to the transfer instruction.
	transfer := system.NewTransferInstruction(1_000_000, sender.PublicKey(), recipient.PublicKey())
	transfer.AccountMetaSlice = append(transfer.AccountMetaSlice, solana.Meta(dontFront))

	// Jito tip: at least 1000 lamports to a random tip account.
	tipAccount := solana.MustPublicKeyFromBase58(tipAccounts[rand.Intn(len(tipAccounts))])
	tip := system.NewTransferInstruction(1_000, sender.PublicKey(), tipAccount)

	recent, err := client.GetLatestBlockhash(ctx, rpc.CommitmentConfirmed)
	if err != nil {
		panic(err)
	}
	tx, err := solana.NewTransaction(
		[]solana.Instruction{transfer.Build(), tip.Build()},
		recent.Value.Blockhash,
		solana.TransactionPayer(sender.PublicKey()),
	)
	if err != nil {
		panic(err)
	}
	tx.Sign(func(solana.PublicKey) *solana.PrivateKey { return &sender.PrivateKey })

	// Sending through Jito submits a real mainnet transaction, so it is off by
	// default and the example prints the signed transaction instead. To send it,
	// sign with a funded mainnet keypair, take the blockhash from a mainnet RPC
	// node, and set SEND_TO_JITO=1.
	if os.Getenv("SEND_TO_JITO") != "1" {
		fmt.Println("Signed transaction (base64):", tx.MustToBase64())
		return
	}

	// Send through Jito's block engine, not a regular RPC node.
	body, _ := json.Marshal(map[string]any{
		"jsonrpc": "2.0",
		"id":      1,
		"method":  "sendTransaction",
		"params":  []any{tx.MustToBase64(), map[string]string{"encoding": "base64"}},
	})
	resp, err := http.Post(jitoEndpoint, "application/json", bytes.NewReader(body))
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	var result struct {
		Result string `json:"result"`
		Error  *struct {
			Message string `json:"message"`
		} `json:"error"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		panic(err)
	}
	if result.Error != nil {
		panic("Jito rejected the transaction: " + result.Error.Message)
	}
	if resp.StatusCode != http.StatusOK {
		panic("Jito returned " + resp.Status)
	}
	fmt.Println("Sent with dontfront protection:", result.Result)
}
